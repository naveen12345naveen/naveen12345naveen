/* TYPING EFFECT */

const words = [
    "ANALYST MODE: ONLINE",
    "DATA → INSIGHT → DECISION",
    "FINANCIAL INTELLIGENCE SYSTEM",
    "AUTOMATION IN PROGRESS"
];

let word = 0;
let letter = 0;
let deleting = false;

const typing =
    document.getElementById("typing");

function typeEffect() {

    const current = words[word];

    if (!deleting) {

        typing.textContent =
            current.substring(0, letter++);

        if (letter > current.length) {

            deleting = true;

            setTimeout(typeEffect, 1200);

            return;
        }

    } else {

        typing.textContent =
            current.substring(0, letter--);

        if (letter < 0) {

            deleting = false;

            letter = 0;

            word++;

            if (word >= words.length) {
                word = 0;
            }

        }
    }

    setTimeout(
        typeEffect,
        deleting ? 40 : 75
    );
}

typeEffect();
